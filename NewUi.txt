-- Utility Functions
local Utility = {}

function Utility:Create(instanceType, properties)
    local instance = Instance.new(instanceType)
    for prop, value in pairs(properties) do
        instance[prop] = value
    end
    return instance
end

function Utility:Tween(instance, properties, duration, style, direction)
    local tween = TweenService:Create(instance, TweenInfo.new(duration or 0.2, style or Enum.EasingStyle.Quart, direction or Enum.EasingDirection.Out), properties)
    tween:Play()
    return tween
end

function Utility:Draggable(frame, parent)
    parent = parent or frame
    local dragging = false
    local dragInput, mousePos, framePos    
    frame.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            mousePos = input.Position
            framePos = parent.Position
            
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    
    frame.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - mousePos
            parent.Position = UDim2.new(framePos.X.Scale, framePos.X.Offset + delta.X, framePos.Y.Scale, framePos.Y.Offset + delta.Y)
        end
    end)
end

function Notifications:Create(title, content, type, duration)

    if not Notifications.Container then
        Notifications.Container = Utility:Create("ScreenGui", {
            Name = "Notifications",

    local colors = {
        info = Config.PrimaryColor,
        success = Config.SuccessColor,
        warning = Config.WarningColor,        error = Config.ErrorColor
    }
    
    local icons = {
        info = "rbxassetid://10889391188",
        success = "rbxassetid://10889384842",
        warning = "rbxassetid://10889384842",
        error = "rbxassetid://10889384842"
    }
    
    local notification = Utility:Create("Frame", {
        Name = "Notification",
        Parent = Notifications.Container,
        BackgroundColor3 = Config.ElementColor,

        Size = UDim2.new(0, 300, 0, 0),

        Position = UDim2.new(1, -320, 1, -10)
    }, 0.4, Enum.EasingStyle.Back)

        Utility:Tween(notification, {
            Position = UDim2.new(1, -20, 1, -10)
        }, 0.4)
        task.wait(0.4)
        notification:Destroy()
    end)

end

                                local currentColor = defaultColor

                    -- Simple color picker modal
                    local hue, sat, val = 0, 1, 1
                    
                    local function hsvToRgb(h, s, v)
                        local r, g, b
                        local i = math.floor(h * 6)
                        local f = h * 6 - i
                        local p = v * (1 - s)
                        local q = v * (1 - f * s)
                        local t = v * (1 - (1 - f) * s)
                        i = i % 6
                        
                        if i == 0 then r, g, b = v, t, p
                        elseif i == 1 then r, g, b = q, v, p
                        elseif i == 2 then r, g, b = p, v, t
                        elseif i == 3 then r, g, b = p, q, v
                        elseif i == 4 then r, g, b = t, p, v
                        elseif i == 5 then r, g, b = v, p, q
                        end
                        
                        return Color3.new(r, g, b)
                    end
                    
                    -- Create simple color picker
                    local pickerGui = Utility:Create("Frame", {
                        Parent = colorPicker,
                        BackgroundColor3 = Color3.fromRGB(30, 30, 35),

                            ColorSequenceKeypoint.new(0.5, Color3.fromRGB(0, 255, 255)),
                            ColorSequenceKeypoint.new(0.67, Color3.fromRGB(0, 0, 255)),
                            ColorSequenceKeypoint.new(0.83, Color3.fromRGB(255, 0, 255)),
                            ColorSequenceKeypoint.new(1, Color3.fromRGB(255, 0, 0))
                        }
                    })
                    
                    local hueButton = Utility:Create("TextButton", {
                        Parent = hueSlider,

                                local x = math.clamp((move.Position.X - hueSlider.AbsolutePosition.X) / hueSlider.AbsoluteSize.X, 0, 1)
                                hue = x
                                currentColor = hsvToRgb(hue, sat, val)
